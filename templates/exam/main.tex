\question[10] Short answers.
\begin{parts}
  \part[5] State the definition of a limit of a sequence.
  \begin{solution}
    $a_n \to a$ if for every $\varepsilon > 0$ there is an $N$ with $|a_n - a| < \varepsilon$ for all $n \ge N$.
  \end{solution}
  \part[5] Is every bounded sequence convergent? Justify your answer.
  \begin{solution}
    No: $a_n = (-1)^n$ is bounded but does not converge.
  \end{solution}
\end{parts}

\question[15] Compute $\displaystyle\int_0^1 x e^{x}\,dx$.
\begin{solution}[3cm]
  By parts: $\int_0^1 x e^x\,dx = [x e^x]_0^1 - \int_0^1 e^x\,dx = e - (e-1) = 1$.
\end{solution}

\question[10] Which of the following is a subspace of $\mathbb{R}^3$?
\begin{choices}
  \choice $\{(x,y,z) : x + y + z = 0\}$
  \CorrectChoice $\{(x,y,z) : x = y\}$
  \choice $\{(x,y,z) : z = 1\}$
  \choice $\{(x,y,z) : xyz = 0\}$
\end{choices}

\question[5] True or false: $\sqrt{2}$ is rational.
\begin{oneparchoices}
  \choice True \CorrectChoice False
\end{oneparchoices}

\question[20] Prove that there are infinitely many primes.
\begin{solution}[5cm]
  Suppose $p_1,\dots,p_n$ were all the primes. Then $N = p_1\cdots p_n + 1$ has a prime
  divisor, which cannot be any $p_i$, a contradiction.
\end{solution}
