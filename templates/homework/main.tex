\begin{problem}{1}
    Show that $\sqrt{2}$ is irrational.
\end{problem}

\begin{solution}
    Suppose $\sqrt{2} = p/q$ with $p, q \in \Z$ and $\gcd(p, q) = 1$. Then $p^2 = 2q^2$,
    so $p$ is even, say $p = 2k$. Then $2k^2 = q^2$, so $q$ is even too, which contradicts
    $\gcd(p, q) = 1$. \qed
\end{solution}
